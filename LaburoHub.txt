-- Obtener el servicio de jugadores
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer

-- En Delta/CoreGui la interfaz es más segura
local coreGui = game:GetService("CoreGui")

-- Limpiar la interfaz si ya existe (para no spammear al ejecutar varias veces)
if coreGui:FindFirstChild("TrollGui") then
    coreGui.TrollGui:Destroy()
end

-- Crear la ScreenGui principal
local screenGui = Instance.new("ScreenGui")
screenGui.Name = "TrollGui"
screenGui.ResetOnSpawn = false
screenGui.Parent = coreGui

-- Crear el fondo de la interfaz
local background = Instance.new("ImageLabel")
background.Size = UDim2.new(0, 300, 0, 200)
background.Position = UDim2.new(0.5, -150, 0.5, -100)
background.BackgroundTransparency = 1
background.Image = "rbxassetid://93021683834741"
background.Parent = screenGui

-- Crear el botón de la habilidad
local killButton = Instance.new("TextButton")
killButton.Size = UDim2.new(0, 200, 0, 50)
killButton.Position = UDim2.new(0.5, -100, 0.5, 0)
killButton.Text = "kill all laburo"
killButton.TextScaled = true
killButton.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
killButton.TextColor3 = Color3.fromRGB(255, 255, 255)
killButton.Font = Enum.Font.FredokaOne
killButton.Parent = background

-- Crear la imagen que aparecerá a pantalla completa
local fullScreenImage = Instance.new("ImageLabel")
fullScreenImage.Size = UDim2.new(1, 0, 1, 0)
fullScreenImage.Position = UDim2.new(0, 0, 0, 0)
fullScreenImage.BackgroundTransparency = 1
fullScreenImage.Image = "rbxassetid://107658241344543"
fullScreenImage.Visible = false
fullScreenImage.ZIndex = 10
fullScreenImage.Parent = screenGui

-- Lista de IDs de sonidos originales
local soundIds = {
    "rbxassetid://117909139728666",
    "rbxassetid://1840254565",
    "rbxassetid://132762282841552",
    "rbxassetid://336771379",
    "rbxassetid://120717906835357",
    "rbxassetid://1300114549451469"
}

-- Función al hacer clic en el botón
killButton.MouseButton1Click:Connect(function()
    -- Mostrar la imagen a pantalla completa
    fullScreenImage.Visible = true
    
    -- Crear y reproducir todos los sonidos en loop
    for _, id in ipairs(soundIds) do
        local sound = Instance.new("Sound")
        sound.SoundId = id
        sound.Looped = true
        sound.Volume = 1
        sound.Parent = screenGui
        sound:Play()
    end
    
    -- Esperar 7 segundos
    task.wait(7)
    
    -- Kickear al jugador con el mensaje
    LocalPlayer:Kick("baneado por falta de laburo ve a laburar vagaso")
end)
